> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let the user edit transactions (every field), categories (name, kind, color, emoji) and accounts (name, type).

**Architecture:** Three `PUT` endpoints that take the same body as the matching create and reuse the existing `*Create` schemas, with services that load the row through the ownership helpers. On the frontend, each create screen's form becomes a component shared by create and edit. Three new `/:id/edit` screens find their record in the lists already loaded. Two PRs into `dev`. The coverage-check docs PR from the spec is written separately once Chris's results are in; it is not part of this plan.

**Tech Stack:** FastAPI, SQLAlchemy 2, Pydantic v2, pytest + httpx TestClient, Postgres 16; React 19 + TypeScript ~5.9, TanStack Query 5, React Router 7, React Hook Form + Zod 4, Tailwind v4, Vitest + RTL + MSW 2.

Spec: `docs/superpowers/specs/2026-10-14-editing-and-plaid-coverage-design.md`.

## Global Constraints

**API:**
- `PUT /api/transactions/{transaction_id}` takes `TransactionCreate`.
- `PUT /api/categories/{category_id}` takes `CategoryCreate`.
- `PUT /api/accounts/{account_id}` takes `AccountCreate`.
- All three return 200 with the existing read schema. No new request schemas.
- API messages, exact: `Transaction not found` (404), `Account not found` (404), `Category not found` (404), `Category name taken` (409), `Account name taken` (409).
- No migration. No single-item GET endpoints.

**Ownership and errors:**
- Every query is scoped to the current user. "Exists but not yours" returns 404, the same as "doesn't exist".
- Services raise domain exceptions. No `HTTPException` or SQLAlchemy exception escapes a service.
- A name clash is caught as `IntegrityError` on commit, then rolled back and raised as `NameTaken`.

**UI copy, exact:**
- Screen titles: `Edit transaction`, `Edit category`, `Edit account`.
- Edit buttons: `Edit transaction`, `Edit account`.
- Submit button on edit: `Save`.
- Kind note: `Changing the kind also changes past months in the summary.`
- Not found: `Transaction not found`, `Category not found`, `Account not found`.
- Category row link label: `Edit <name>`.

**Frontend routes:** `/transactions/:id/edit`, `/categories/:id/edit`, `/accounts/:id/edit`.

**Behaviour:**
- The "+ New category" shortcut appears on the create transaction screen only, never on edit.
- Sign convention: `amount` is signed; negative is money out. The form's amount field is always positive, and the Money out / Money in toggle supplies the sign. Amounts stay strings, with no float math.
- The existing create-screen tests must pass unchanged after the forms are extracted. Do not edit them to make them pass.

**Tooling:**
- `frontend/src/api/schema.d.ts` is generated only (`npm run gen:api` with the backend on port 8010). Never hand-edit it.
- Backend tooling runs from `backend/` with `uv run ...`. Ruff must be clean: `uv run ruff check . && uv run ruff format --check .`.
- Frontend: use the existing components (Screen, Card, Field, Button), give inputs `inputClass`, and keep touch targets at least `min-h-11`.

```tsx
import { zodResolver } from "@hookform/resolvers/zod";
import { useForm } from "react-hook-form";
import { z } from "zod";
import type { Account } from "../api/accounts";
import type { Category } from "../api/categories";
import { useCreateTransaction } from "../api/transactions";
import { Button } from "../components/Button";
import { Field } from "../components/Field";
import { Segmented } from "../components/Segmented";
import { inputClass } from "../components/styles";
import { accountTypeIcon } from "../lib/accountTypes";
import { CATEGORY_KINDS, kindLabel } from "../lib/categoryKinds";
import { addDays, localToday } from "../lib/dates";

const DIRECTIONS = [
  { value: "out", label: "Money out" },
  { value: "in", label: "Money in" },
] as const;

// Amounts stay strings end to end (NUMERIC(12,2) on the backend); we only
// validate the shape here and never do float math on them. The field is always
// positive: the direction supplies the sign (money out is negative).
const amountMessage = "Enter an amount greater than 0 with up to 2 decimals";
const schema = z.object({
  direction: z.enum(["out", "in"]),
  account_id: z.string().min(1, "Pick an account"),
  category_id: z.string().min(1, "Pick a category"),
  amount: z
    .string()
    .regex(/^\d{1,10}(\.\d{1,2})?$/, amountMessage)
    .refine((v) => Number(v) > 0, amountMessage),
  occurred_on: z
    .string()
    .min(1, "Date is required")
    // Backend allows one day of slack for timezone skew; mirror it.
    .refine((v) => v <= addDays(localToday(), 1), "Date cannot be in the future"),
  note: z.string().trim().max(255, "Note must be at most 255 characters"),
});
export type TransactionFormValues = z.infer<typeof schema>;

// Form values carried to the new-category screen and back, so nothing typed is lost.
export type TransactionDraft = Partial<TransactionFormValues>;

// Grouped by kind, spending first: the order the category select shows them in.
export function groupCategories(categories: Category[]) {
  return CATEGORY_KINDS.map((kind) => ({ kind, items: categories.filter((c) => c.kind === kind) }));
}

type Props = {
  accounts: Account[];
  categories: Category[];
  defaultValues: TransactionFormValues;
  onSaved: () => void;
  // Create only: opens the new-category screen with what's typed so far.
  onNewCategory?: (draft: TransactionDraft) => void;
};

export function TransactionForm({ accounts, categories, defaultValues, onSaved, onNewCategory }: Props) {
  const groups = groupCategories(categories);
  const create = useCreateTransaction();
  const {
    register,
    handleSubmit,
    getValues,
    formState: { errors, isSubmitting },
  } = useForm<TransactionFormValues>({ resolver: zodResolver(schema), defaultValues });

  const onSubmit = handleSubmit(async (values) => {
    try {
      await create.mutateAsync({
        account_id: Number(values.account_id),
        category_id: Number(values.category_id),
        amount: values.direction === "out" ? `-${values.amount}` : values.amount,
        occurred_on: values.occurred_on,
        note: values.note || null,
      });
      onSaved();
    } catch {
      // Shown below via create.error
    }
  });

  return (
    <form onSubmit={onSubmit} noValidate className="flex flex-col gap-4">
      <Segmented legend="Direction" hideLegend options={DIRECTIONS} field={register("direction")} />
      <Field label="Amount" htmlFor="amount" error={errors.amount?.message}>
        <input id="amount" inputMode="decimal" placeholder="0.00" className={`${inputClass} text-2xl font-bold`} {...register("amount")} />
      </Field>
      <Field label="Account" htmlFor="account_id" error={errors.account_id?.message}>
        <select id="account_id" className={inputClass} {...register("account_id")}>
          {accounts.length > 1 && <option value="">Choose an account</option>}
          {accounts.map((a) => (
            <option key={a.id} value={a.id}>
              {accountTypeIcon(a.type)} {a.name}
            </option>
          ))}
        </select>
      </Field>
      <Field label="Category" htmlFor="category_id" error={errors.category_id?.message}>
        <select id="category_id" className={inputClass} {...register("category_id")}>
          {groups
            .filter((g) => g.items.length > 0)
            .map((g) => (
              <optgroup key={g.kind} label={kindLabel(g.kind)}>
                {g.items.map((c) => (
                  <option key={c.id} value={c.id}>
                    {c.icon} {c.name}
                  </option>
                ))}
              </optgroup>
            ))}
        </select>
        {onNewCategory && (
          <button
            type="button"
            onClick={() => onNewCategory(getValues())}
            className="min-h-11 self-start text-sm font-semibold text-accent"
          >
            + New category
          </button>
        )}
      </Field>
      <Field label="Date" htmlFor="occurred_on" error={errors.occurred_on?.message}>
        <input id="occurred_on" type="date" className={inputClass} {...register("occurred_on")} />
      </Field>
      <Field label="Note" htmlFor="note" error={errors.note?.message}>
        <input id="note" placeholder="Optional" className={inputClass} {...register("note")} />
      </Field>
      {create.isError && (
        <p role="alert" className="text-sm text-danger">
          {create.error.message}
        </p>
      )}
      <Button type="submit" disabled={isSubmitting} className="mt-2 w-full">
        Add transaction
      </Button>
    </form>
  );
}
```

```tsx
import { zodResolver } from "@hookform/resolvers/zod";
import { useForm } from "react-hook-form";

import { Field } from "../components/Field";
import { inputClass } from "../components/styles";
import { ACCOUNT_TYPES, accountTypeIcon, accountTypeLabel } from "../lib/accountTypes";

// Mirrors the backend AccountCreate schema.
const schema = z.object({
  name: z.string().trim().min(1, "Name is required").max(50, "Name must be at most 50 characters"),
  type: z.enum(ACCOUNT_TYPES as [AccountType, ...AccountType[]]),
});
type FormValues = z.infer<typeof schema>;

type Props = {
  // Set when editing: the form starts from it and saves with PUT.
  account?: Account;
  onSaved: () => void;
};

export function AccountForm({ account, onSaved }: Props) {

    defaultValues: account ? { name: account.name, type: account.type } : { name: "", type: "checking" },

    try {
      if (account) await update.mutateAsync({ id: account.id, body: values });

      // Shown below via save.error
    }
  });

        <legend className="mb-2 text-sm font-medium">Type</legend>
```
