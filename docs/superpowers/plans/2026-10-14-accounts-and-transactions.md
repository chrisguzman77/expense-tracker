# Accounts and Transactions (piece 2a) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add accounts, reshape expenses into signed transactions that belong to an account, and give categories a kind (spending, income, transfer) so only spending is summed.

**Architecture:** Three PRs into `dev`, each with one Alembic migration. PR 1 adds an `accounts` table, API and Accounts tab. PR 2 renames `expenses` to `transactions` (with a data migration tested against a real database), makes amounts signed and links each transaction to an account. PR 3 adds `categories.kind` and filters the summary to spending.

**Tech Stack:** FastAPI, SQLAlchemy 2, Alembic, Pydantic v2, pytest + httpx TestClient, Postgres 16; React 19 + TypeScript ~5.9, TanStack Query 5, React Router 7, React Hook Form + Zod 4, Tailwind v4, Vitest + RTL + MSW 2.

Spec: `docs/superpowers/specs/2026-10-14-accounts-and-transactions-design.md`.

## Global Constraints

- Sign convention: `amount` is signed; **negative is money out, positive is money in**; zero is rejected (422). Amounts travel as decimal strings ("-12.50").
- Account types, exactly: `checking`, `savings`, `credit_card`, `cash`. UI labels: Checking, Savings, Credit card, Cash. Icons: 🏦, 🐷, 💳, 💵.
- Category kinds, exactly: `spending`, `income`, `transfer`; default `spending`. UI labels: Spending, Income, Transfer.
- API messages, exact: `Account name taken` (409), `Account not found` (404), `Account has transactions` (409), `Category not found` (404), `Category has transactions` (409), `Transaction not found` (404).
- UI copy, exact: `No accounts yet.`, `Add an account first.`, `This account still has transactions; delete them first.`, `Pick an account`, `Money out`, `Money in`.
- Alembic revision ids (pass with `--rev-id`): `5e2b7c41a9d3` create accounts; `8f4d2a6c1e70` reshape; `b37e95d0c2a4` category kind.
- Ownership: every query is scoped to the current user; "exists but not yours" returns 404, same as "doesn't exist".
- `backend/app/models/__init__.py` must import every model module. `backend/alembic/env.py`'s `import app.models` is load-bearing; never remove it.
- Read every Alembic migration before `upgrade`. An autogenerate that touches tables you didn't change means model imports are broken: stop and report.
- Backend tooling runs from `backend/` with `uv run ...`. Ruff must be clean (`uv run ruff check . && uv run ruff format --check .`).
- `frontend/src/api/schema.d.ts` is generated only (`npm run gen:api` with the backend on port 8010); never hand-edit it.
- Frontend: existing components (Screen, Card, Field, Button, GradientHeader), inputs use `inputClass` (16px text), touch targets at least `min-h-11`.
- Never read or edit any `.env` or `.env.example` file.

Create `backend/app/services/accounts.py`:

```python
from sqlalchemy import select
from sqlalchemy.exc import IntegrityError
from sqlalchemy.orm import Session

from ..models.account import Account


class NotFound(Exception): ...


class NameTaken(Exception): ...


def list_accounts(db: Session, user_id: int) -> list[Account]:
    return list(
        db.scalars(
            select(Account).where(Account.user_id == user_id).order_by(Account.name)
        )
    )


def create_account(
    db: Session, user_id: int, name: str, account_type: str
) -> Account:
    account = Account(user_id=user_id, name=name, type=account_type)
    db.add(account)
    try:
        db.commit()
    except IntegrityError:
        db.rollback()
        raise NameTaken()
    db.refresh(account)
    return account


def get_owned_account(db: Session, user_id: int, account_id: int) -> Account:
    account = db.get(Account, account_id)
    if account is None or account.user_id != user_id:
        raise NotFound()
    return account


def delete_account(db: Session, user_id: int, account_id: int) -> None:
    account = get_owned_account(db, user_id, account_id)
    db.delete(account)
    db.commit()
```

```python
from fastapi import APIRouter, Depends, HTTPException, status
from sqlalchemy.orm import Session

from ..auth import get_current_user
from ..database import get_db
from ..models.user import User
from ..schemas.account import AccountCreate, AccountRead
from ..services import accounts as svc

router = APIRouter(prefix="/api/accounts", tags=["accounts"])


@router.get("", response_model=list[AccountRead])
def list_accounts(
    user: User = Depends(get_current_user), db: Session = Depends(get_db)
):
    return svc.list_accounts(db, user.id)


@router.post("", response_model=AccountRead, status_code=201)
def create_account(
    body: AccountCreate,
    user: User = Depends(get_current_user),
    db: Session = Depends(get_db),
):
    try:
        return svc.create_account(db, user.id, body.name, body.type)
    except svc.NameTaken:
        raise HTTPException(status.HTTP_409_CONFLICT, "Account name taken")


@router.delete("/{account_id}", status_code=204)
def delete_account(
    account_id: int,
    user: User = Depends(get_current_user),
    db: Session = Depends(get_db),
):
    try:
        svc.delete_account(db, user.id, account_id)
    except svc.NotFound:
        raise HTTPException(status.HTTP_404_NOT_FOUND, "Account not found")
```

```ts
import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import { api, unwrap, unwrapVoid } from "./client";
import type { components } from "./schema";

export type Account = components["schemas"]["AccountRead"];
export type AccountCreate = components["schemas"]["AccountCreate"];
export type AccountType = Account["type"];

export const accountsQueryKey = ["accounts"] as const;

export function useAccounts() {
  return useQuery({
    queryKey: accountsQueryKey,
    queryFn: () => unwrap(api.GET("/api/accounts")),
  });
}

export function useCreateAccount() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: (body: AccountCreate) => unwrap(api.POST("/api/accounts", { body })),
    onSuccess: () => queryClient.invalidateQueries({ queryKey: accountsQueryKey }),
  });
}

export function useDeleteAccount() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: (id: number) =>
      unwrapVoid(api.DELETE("/api/accounts/{account_id}", { params: { path: { account_id: id } } })),
    onSuccess: () => queryClient.invalidateQueries({ queryKey: accountsQueryKey }),
  });
}
```

```tsx
import { useState } from "react";
import { useNavigate, useParams } from "react-router-dom";
import { useAccounts, useDeleteAccount } from "../api/accounts";

import { AccountIcon } from "../components/AccountIcon";
import { Button } from "../components/Button";

import { Screen } from "../components/Screen";
import { accountTypeLabel } from "../lib/accountTypes";

// Reads from the already-loaded account list; the API has no single-account GET.
export function AccountDetailPage() {
  const { id } = useParams();
  const navigate = useNavigate();
  const accounts = useAccounts();
  const remove = useDeleteAccount();
  const [confirming, setConfirming] = useState(false);
  const account = accounts.data?.find((a) => a.id === Number(id));

  const removeError =
    remove.error instanceof ApiError && remove.error.status === 409
      ? "This account still has transactions; delete them first."
      : remove.error?.message;

    try {
      await remove.mutateAsync(accountId);
      navigate("/accounts", { replace: true });
    } catch {
      // Shown below via removeError
    }
  }

  function onCancel() {
    setConfirming(false);
    remove.reset();
  }

  return (
    <Screen title="Account" backTo="/accounts">
      {accounts.isPending && <p className="text-muted">Loading…</p>}
      {accounts.isError && (
        <p role="alert" className="text-sm text-danger">
          {accounts.error.message}
        </p>
      )}

              {removeError}
            </p>
          )}

            <Card className="flex flex-col gap-3 p-4">

                </Button>
                <Button variant="danger" onClick={() => onDelete(account.id)} disabled={remove.isPending}>
                  Delete
                </Button>
              </div>

            <Button variant="danger" onClick={() => setConfirming(true)} className="w-full">
              Delete account
            </Button>

        </div>
```

```python
def list_transactions(
    db: Session,
    user_id: int,
    start: date | None,
    end: date | None,
    account_id: int | None,
) -> list[Transaction]:
    q = (
        select(Transaction)
        .where(Transaction.user_id == user_id)
        .options(selectinload(Transaction.category), selectinload(Transaction.account))
    )
    if account_id is not None:
        # Raises accounts.NotFound for someone else's account, rather than
        # answering with an empty list that hides a wrong id.
        accounts.get_owned_account(db, user_id, account_id)
        q = q.where(Transaction.account_id == account_id)
    if start:
        q = q.where(Transaction.occurred_on >= start)
    if end:
        q = q.where(Transaction.occurred_on <= end)
    return list(
        db.scalars(q.order_by(Transaction.occurred_on.desc(), Transaction.id.desc()))
    )


def create_transaction(db: Session, user_id: int, data) -> Transaction:

    _owned_category(db, user_id, data.category_id)
    txn = Transaction(user_id=user_id, **data.model_dump())

    db.refresh(txn)

def summary_by_category(db: Session, user_id: int, year: int, month: int):
    start = date(year, month, 1)
    end = date(year + (month == 12), (month % 12) + 1, 1)
    # Money out is negative, so spending is minus the sum: refunds reduce it.
    spent = -func.sum(Transaction.amount)
    q = (
        select(
            Category.id,
            Category.name,
            Category.color,
            Category.icon,
            func.coalesce(spent, 0).label("total"),
        )
        # Date range lives in the ON clause so categories with no transactions
        # this month survive the outer join (a WHERE filter would drop them).
        .join(
            Transaction,
            (Transaction.category_id == Category.id)
            & (Transaction.occurred_on >= start)
            & (Transaction.occurred_on < end),
            isouter=True,
        )
        .where(Category.user_id == user_id)
        .group_by(Category.id)
        .order_by(spent.desc().nulls_last())
    )
    return [
        {"category_id": r[0], "name": r[1], "color": r[2], "icon": r[3], "total": r[4]}
        for r in db.execute(q)
    ]
```

```python
def upgrade() -> None:
    """Expenses become signed transactions that belong to an account."""
    # Postgres keeps the old names of the sequence, constraints and index when
    # a table is renamed, so rename them too.
    op.rename_table("expenses", "transactions")
    op.execute("ALTER SEQUENCE expenses_id_seq RENAME TO transactions_id_seq")
    op.execute(
        "ALTER TABLE transactions RENAME CONSTRAINT expenses_pkey TO transactions_pkey"
    )
    op.execute(
        "ALTER TABLE transactions RENAME CONSTRAINT expenses_user_id_fkey "
        "TO transactions_user_id_fkey"
    )
    op.execute(
        "ALTER TABLE transactions RENAME CONSTRAINT expenses_category_id_fkey "
        "TO transactions_category_id_fkey"
    )
    op.alter_column("transactions", "spent_on", new_column_name="occurred_on")
    op.execute(
        "ALTER INDEX ix_expenses_user_spent_on RENAME TO ix_transactions_user_occurred_on"
    )

    # Every existing row moves into an account named "Cash". Accounts shipped
    # first, so a user may already have one by that name: reuse it.
    op.add_column("transactions", sa.Column("account_id", sa.Integer(), nullable=True))
    op.execute(
        "INSERT INTO accounts (user_id, name, type) "
        "SELECT DISTINCT user_id, 'Cash', 'cash' FROM transactions "
        "ON CONFLICT (user_id, name) DO NOTHING"
    )
    op.execute(
        "UPDATE transactions AS t SET account_id = a.id FROM accounts AS a "
        "WHERE a.user_id = t.user_id AND a.name = 'Cash'"
    )
    op.alter_column("transactions", "account_id", nullable=False)
    op.create_foreign_key(
        "transactions_account_id_fkey",
        "transactions",
        "accounts",
        ["account_id"],
        ["id"],
        ondelete="RESTRICT",
    )
    op.create_index(
        "ix_transactions_account_occurred_on",
        "transactions",
        ["account_id", "occurred_on"],
    )

    # Every existing row is money out, which is now negative.
    op.execute("UPDATE transactions SET amount = -amount")


def downgrade() -> None:
    """Back to expenses. Accounts are user data once created and stay; the
    accounts migration's downgrade drops that table. Lossy once money-in rows
    exist: they come back as negative expenses, which the old API rejects."""
    op.execute("UPDATE transactions SET amount = -amount")
    op.drop_index("ix_transactions_account_occurred_on", table_name="transactions")
    op.drop_constraint(
        "transactions_account_id_fkey", "transactions", type_="foreignkey"
    )
    op.drop_column("transactions", "account_id")

    op.execute(
        "ALTER INDEX ix_transactions_user_occurred_on RENAME TO ix_expenses_user_spent_on"
    )
    op.alter_column("transactions", "occurred_on", new_column_name="spent_on")
    op.execute(
        "ALTER TABLE transactions RENAME CONSTRAINT transactions_category_id_fkey "
        "TO expenses_category_id_fkey"
    )
    op.execute(
        "ALTER TABLE transactions RENAME CONSTRAINT transactions_user_id_fkey "
        "TO expenses_user_id_fkey"
    )
    op.execute(
        "ALTER TABLE transactions RENAME CONSTRAINT transactions_pkey TO expenses_pkey"
    )
    op.execute("ALTER SEQUENCE transactions_id_seq RENAME TO expenses_id_seq")
    op.rename_table("transactions", "expenses")
```

```tsx
// Money out (negative) reads like a price, with no minus sign; money in gets a
// plus sign and the income color, so the two never look alike.
export function TransactionAmount({ amount, className = "" }: { amount: string; className?: string }) {
  const cents = toCents(amount);
  if (cents > 0) {
    return <span className={`tabular-nums text-income ${className}`}>+{formatCents(cents)}</span>;
  }
```

```ts
import { type QueryClient, useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import { api, unwrap, unwrapVoid } from "./client";
import type { components } from "./schema";
import { summaryQueryKey } from "./summary";

export type Transaction = components["schemas"]["TransactionRead"];
export type TransactionCreate = components["schemas"]["TransactionCreate"];

export const transactionsQueryKey = ["transactions"] as const;

// Monthly totals are computed from transactions, so they go stale together.
function invalidateTransactionData(queryClient: QueryClient) {
  return Promise.all([
    queryClient.invalidateQueries({ queryKey: transactionsQueryKey }),
    queryClient.invalidateQueries({ queryKey: summaryQueryKey }),
  ]);
}

export function useTransactions() {
  return useQuery({
    queryKey: transactionsQueryKey,
    queryFn: () => unwrap(api.GET("/api/transactions")),
  });
}

export function useCreateTransaction() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: (body: TransactionCreate) => unwrap(api.POST("/api/transactions", { body })),
    onSuccess: () => invalidateTransactionData(queryClient),
  });
}

export function useDeleteTransaction() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: (id: number) =>
      unwrapVoid(
        api.DELETE("/api/transactions/{transaction_id}", { params: { path: { transaction_id: id } } }),
      ),
    onSuccess: () => invalidateTransactionData(queryClient),
```

```ts
import type { components } from "../api/schema";

export type CategoryKind = components["schemas"]["CategoryRead"]["kind"];

// A Record keyed by the generated API union: tsc fails if the lists drift apart.
const LABELS: Record<CategoryKind, string> = {

export const CATEGORY_KINDS = Object.keys(LABELS) as CategoryKind[];

export function kindLabel(kind: CategoryKind): string {
  return LABELS[kind];
}
```
